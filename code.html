const JSON_HEADERS = { "content-type": "application/json; charset=utf-8" };

function json(data, status = 200, extraHeaders = {}) {
  return new Response(JSON.stringify(data, null, 2), {
    status,
    headers: { ...JSON_HEADERS, ...extraHeaders },
  });
}

function corsHeaders(request, env) {
  const origin = request.headers.get("Origin") || "";
  const allowed = env.ALLOWED_ORIGIN || "*";
  const allowOrigin = allowed === "*" || origin === allowed ? (origin || "*") : allowed;

  return {
    "access-control-allow-origin": allowOrigin,
    "access-control-allow-methods": "GET,POST,PATCH,DELETE,OPTIONS",
    "access-control-allow-headers": "Content-Type",
    "access-control-max-age": "86400",
    "vary": "Origin",
  };
}

function cleanText(value, max = 180) {
  return String(value ?? "").trim().slice(0, max);
}

function validStage(stage) {
  return ["New", "Qualified", "Proposal", "Won", "Lost"].includes(stage);
}

function leadPayload(body) {
  const payload = {
    name: cleanText(body.name, 100),
    company: cleanText(body.company, 120),
    email: cleanText(body.email, 160),
    source: cleanText(body.source || "Website", 80),
    value: Number(body.value || 0),
    stage: cleanText(body.stage || "New", 30),
    owner: cleanText(body.owner || "Adrian", 80),
  };

  if (!payload.name || !payload.company || !payload.email) {
    return { error: "name, company and email are required" };
  }
  if (!payload.email.includes("@")) return { error: "email is invalid" };
  if (!Number.isFinite(payload.value) || payload.value < 0 || payload.value > 10000000) {
    return { error: "value must be a positive number" };
  }
  if (!validStage(payload.stage)) return { error: "invalid stage" };

  return { payload };
}

async function listLeads(env, url) {
  const stage = cleanText(url.searchParams.get("stage") || "", 30);
  const q = cleanText(url.searchParams.get("q") || "", 100).toLowerCase();

  const result = await env.DB.prepare(
    `SELECT id, name, company, email, source, value, stage, owner, created_at, updated_at
     FROM leads
     ORDER BY id DESC
     LIMIT 100`
  ).all();

  let rows = result.results || [];
  if (stage) rows = rows.filter((row) => row.stage === stage);
  if (q) {
    rows = rows.filter((row) =>
      `${row.name} ${row.company} ${row.email}`.toLowerCase().includes(q)
    );
  }

  const pipelineValue = rows
    .filter((row) => row.stage !== "Lost")
    .reduce((sum, row) => sum + Number(row.value || 0), 0);

  return {
    leads: rows,
    meta: {
      count: rows.length,
      pipeline_value: pipelineValue,
      qualified: rows.filter((row) => row.stage === "Qualified").length,
      won: rows.filter((row) => row.stage === "Won").length,
    },
  };
}

async function createLead(request, env) {
  const body = await request.json().catch(() => null);
  if (!body) return { error: "invalid JSON body", status: 400 };

  const parsed = leadPayload(body);
  if (parsed.error) return { error: parsed.error, status: 400 };

  const p = parsed.payload;
  const result = await env.DB.prepare(
    `INSERT INTO leads (name, company, email, source, value, stage, owner)
     VALUES (?, ?, ?, ?, ?, ?, ?)`
  ).bind(p.name, p.company, p.email, p.source, p.value, p.stage, p.owner).run();

  // Keep the public demo bounded.
  await env.DB.prepare(`
    DELETE FROM leads
    WHERE id NOT IN (SELECT id FROM leads ORDER BY id DESC LIMIT 100)
  `).run();

  const lead = await env.DB.prepare(
    `SELECT id, name, company, email, source, value, stage, owner, created_at, updated_at
     FROM leads WHERE id = ?`
  ).bind(result.meta.last_row_id).first();

  return { lead, status: 201 };
}

async function updateLead(request, env, id) {
  const body = await request.json().catch(() => null);
  if (!body) return { error: "invalid JSON body", status: 400 };

  const current = await env.DB.prepare(`SELECT * FROM leads WHERE id = ?`).bind(id).first();
  if (!current) return { error: "lead not found", status: 404 };

  const next = {
    name: cleanText(body.name ?? current.name, 100),
    company: cleanText(body.company ?? current.company, 120),
    email: cleanText(body.email ?? current.email, 160),
    source: cleanText(body.source ?? current.source, 80),
    value: Number(body.value ?? current.value),
    stage: cleanText(body.stage ?? current.stage, 30),
    owner: cleanText(body.owner ?? current.owner, 80),
  };

  const parsed = leadPayload(next);
  if (parsed.error) return { error: parsed.error, status: 400 };

  const p = parsed.payload;
  await env.DB.prepare(
    `UPDATE leads
     SET name=?, company=?, email=?, source=?, value=?, stage=?, owner=?,
         updated_at=CURRENT_TIMESTAMP
     WHERE id=?`
  ).bind(p.name, p.company, p.email, p.source, p.value, p.stage, p.owner, id).run();

  const lead = await env.DB.prepare(
    `SELECT id, name, company, email, source, value, stage, owner, created_at, updated_at
     FROM leads WHERE id = ?`
  ).bind(id).first();

  return { lead, status: 200 };
}

async function deleteLead(env, id) {
  const result = await env.DB.prepare(`DELETE FROM leads WHERE id = ?`).bind(id).run();
  if (!result.meta.changes) return { error: "lead not found", status: 404 };
  return { ok: true, status: 200 };
}

export default {
  async fetch(request, env) {
    const cors = corsHeaders(request, env);
    if (request.method === "OPTIONS") return new Response(null, { headers: cors });

    const url = new URL(request.url);
    const path = url.pathname.replace(/\/+$/, "") || "/";

    try {
      if (path === "/" || path === "/health") {
        return json({
          ok: true,
          service: "adrian-portfolio-api",
          database: "cloudflare-d1",
          timestamp: new Date().toISOString(),
        }, 200, cors);
      }

      if (path === "/api/leads" && request.method === "GET") {
        return json(await listLeads(env, url), 200, cors);
      }

      if (path === "/api/leads" && request.method === "POST") {
        const result = await createLead(request, env);
        return json(result.error ? { error: result.error } : { lead: result.lead }, result.status, cors);
      }

      const match = path.match(/^\/api\/leads\/(\d+)$/);
      if (match && request.method === "PATCH") {
        const result = await updateLead(request, env, Number(match[1]));
        return json(result.error ? { error: result.error } : { lead: result.lead }, result.status, cors);
      }

      if (match && request.method === "DELETE") {
        const result = await deleteLead(env, Number(match[1]));
        return json(result.error ? { error: result.error } : { ok: true }, result.status, cors);
      }

      return json({ error: "not found" }, 404, cors);
    } catch (error) {
      console.error(error);
      return json({ error: "internal server error" }, 500, cors);
    }
  },
};

CREATE TABLE leads (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  name TEXT NOT NULL,
  company TEXT NOT NULL,
  email TEXT NOT NULL,
  source TEXT NOT NULL DEFAULT 'Website',
  value REAL NOT NULL DEFAULT 0 CHECK (value >= 0),
  stage TEXT NOT NULL DEFAULT 'New'
    CHECK (stage IN ('New','Qualified','Proposal','Won','Lost')),
  owner TEXT NOT NULL DEFAULT 'Adrian',
  created_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at TEXT NOT NULL DEFAULT CURRENT_TIMESTAMP
);

CREATE INDEX idx_leads_stage ON leads(stage);
CREATE INDEX idx_leads_created_at ON leads(created_at);

      next_action: 'route_to_sales'
    };

  }
